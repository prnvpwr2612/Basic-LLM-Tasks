def load_captions(filename):
 captions = {}
 with open(filename, 'r') as f:
 for line in f:
 tokens = line.split()
 image_id, caption = tokens[0].split('#')[0], tokens[1:]
 caption = " ".join(caption)
 if image_id not in captions:
 captions[image_id] = []
 captions[image_id].append(caption)
 return captions

class ImageCaptioningDataset(Dataset):
 def __init__(self, image_caption_pairs, feature_extractor, tokenizer, max_length=50):
 self.image_caption_pairs = image_caption_pairs
 self.feature_extractor = feature_extractor
 self.tokenizer = tokenizer
 self.max_length = max_length
 def __len__(self):
 return len(self.image_caption_pairs)
 def __getitem__(self, idx):

 try:
 image = Image.open(image_path).convert("RGB")

 input_ids = inputs.input_ids.squeeze(0)

 return {"pixel_values": pixel_values, "input_ids": input_ids, "attention_mask": attention_mask}

class MultimodalCaptioningModel(torch.nn.Module):
 def __init__(self):
 super(MultimodalCaptioningModel, self).__init__()
 self.vision_encoder = ViTModel.from_pretrained('google/vit-base-patch16-224-in21k')
 for param in self.vision_encoder.parameters():
 param.requires_grad = False
 self.language_model = GPT2LMHeadModel.from_pretrained('gpt2')
 self.vision_projection = torch.nn.Linear(self.vision_encoder.config.hidden_size,
self.language_model.config.hidden_size)
 def forward(self, pixel_values, input_ids, attention_mask):
 with torch.no_grad():
 image_features = self.vision_encoder(pixel_values).last_hidden_state[:, 0, :]
 projected_image_features = self.vision_projection(image_features)
 text_embeddings = self.language_model.transformer.wte(input_ids)
 projected_image_features_expanded = projected_image_features.unsqueeze(1)
 combined_embeddings = torch.cat([projected_image_features_expanded, text_embeddings], dim=1)
 image_attention_mask = torch.ones(projected_image_features_expanded.size(0), 1,
dtype=torch.long).to(attention_mask.device)
 combined_attention_mask = torch.cat([image_attention_mask, attention_mask], dim=1)
 image_labels = torch.full((projected_image_features.size(0), 1), -100,
dtype=torch.long).to(input_ids.device)
 combined_labels = torch.cat([image_labels, input_ids], dim=1)
 outputs = self.language_model(
 inputs_embeds=combined_embeddings, attention_mask=combined_attention_mask,
labels=combined_labels
 )
 return outputs
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

def generate_caption_for_image(model, image_path, feature_extractor, tokenizer, device, max_length=50):
 """Generates a caption for a given image using the trained model."""
 model.eval() # Ensure the model is in evaluation mode
 try:
 image = Image.open(image_path).convert("RGB")
 except (IOError, OSError):
 print(f"Error: Could not load image at {image_path}.")
 return "Could not generate a caption because the image could not be loaded."
 pixel_values = feature_extractor(images=image, return_tensors="pt").pixel_values.to(device)
 with torch.no_grad():
 # Get image embeddings from the vision encoder and project them
 image_features = model.vision_encoder(pixel_values).last_hidden_state[:, 0, :]
 projected_image_features = model.vision_projection(image_features)
 inputs_embeds = projected_image_features.unsqueeze(1)
 # Use the language model's generate method to create the caption
 output_sequences = model.language_model.generate(
 inputs_embeds=inputs_embeds,
 max_length=max_length,
 num_beams=5, # Use beam search for higher quality results
 no_repeat_ngram_size=2,
 early_stopping=True,
 pad_token_id=tokenizer.eos_token_id
 )
 generated_caption = tokenizer.decode(output_sequences[0], skip_special_tokens=True)
 return generated_caption.strip()
